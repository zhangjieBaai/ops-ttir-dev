"""Prove tutorial source and frozen TTIR share the native downstream pipeline."""

import argparse
import inspect
import os
import platform
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path

import torch
from common import (
    ROOT,
    SEED_SHAPE,
    allocate,
    compile_ir,
    dump_kernel,
    frozen,
    gate,
    launch,
    launch_source,
    save_json,
    sha256,
)
from environment import verify_distribution_files
from export_baseline import verify_extraction
from triton._C import libtriton
from triton._C.libtriton import ir
from triton.backends.compiler import Language
from triton.backends.nvidia.compiler import CUDABackend, get_ptxas


def snapshot():
    return subprocess.check_output(
        [
            "nvidia-smi",
            "--query-gpu=index,uuid,name,driver_version,pci.bus_id,clocks.sm,clocks.mem,"
            "temperature.gpu,power.draw,power.limit,utilization.gpu,memory.used",
            "--format=csv",
        ],
        text=True,
    )


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    assert not args.output.exists()
    target = gate()
    environment = verify_distribution_files()
    verify_extraction()
    config = frozen()
    backend = CUDABackend(target)
    options = backend.parse_options(config["options"])
    stages = {}
    backend.add_stages(stages, options, Language.TRITON)
    assert list(stages) == ["ttir", "ttgir", "llir", "ptx", "cubin"]
    torch.manual_seed(42)
    tensors = allocate(*SEED_SHAPE)
    source = launch_source(tensors, config)
    source_info = dump_kernel(source, "tutorial")
    assert source.asm["ttir"] == (ROOT / "ir/baseline.ttir").read_text()
    reference = tensors[2].clone()
    baseline = compile_ir()
    actual_options = [
        {key: ck.metadata._asdict()[key] for key in vars(options)}
        for ck in [source, baseline]
    ]
    assert actual_options[0] == actual_options[1]
    assert source.metadata.target == baseline.metadata.target == target
    identical = {
        stage: source.asm[stage] == baseline.asm[stage]
        for stage in ["llir", "ptx", "cubin"]
    }
    assert all(identical.values()), identical
    context = ir.context()
    ir.load_dialects(context)
    backend.load_dialects(context)
    normalized = []
    for ck in [source, baseline]:
        with tempfile.NamedTemporaryFile(mode="w", suffix=".ttgir") as f:
            f.write(ck.asm["ttgir"])
            f.flush()
            normalized.append(str(ir.parse_mlir_module(f.name, context)))
    assert normalized[0] == normalized[1]
    tensors[3].fill_(float("nan"))
    launch(baseline, tensors, config)
    torch.testing.assert_close(tensors[2], reference, rtol=0, atol=0)
    # Freeze and inspect the actual entry assumptions, not only Python intent.
    entry = (
        (ROOT / "ir/baseline.ttir")
        .read_text()
        .split("tt.func public @", 1)[1]
        .split(" {\n", 1)[0]
    )
    assert entry.count("%arg") == 9, entry
    assert entry.count("tt.divisibility = 16") == 9, entry
    torch.cuda.synchronize()
    libraries = sorted(
        {
            line.split()[-1]
            for line in Path("/proc/self/maps").read_text().splitlines()
            if ".so" in line
        }
    )
    torch_components = [
        p for p in libraries if any(s in p for s in ["libtorch", "libc10", "libcudart"])
    ]
    assert all(
        p.startswith(sys.prefix + "/") for p in torch_components
    ), torch_components
    assert not any(
        "flagtree" in p.lower() or "flagprism" in p.lower() for p in libraries
    )
    ptxas = get_ptxas(90).path
    tool_paths = [
        inspect.getfile(CUDABackend),
        libtriton.__file__,
        ptxas,
        ROOT / "environment.py",
    ]
    tool_paths += [p for _, p in options.extern_libs]
    result = dict(
        passed=True,
        experiment=config["experiment"],
        environment=environment,
        python=platform.python_version(),
        torch=torch.__version__,
        torch_path=torch.__file__,
        target=str(target),
        actual_options=actual_options[0],
        baseline_sha256=config["ttir_sha256"],
        config_sha256=sha256(ROOT / "baseline.json"),
        source_sha256=config["source_sha256"],
        contract_sha256=config["contract_sha256"],
        entry_signature=entry,
        tutorial_vs_file_byte_identical=identical,
        ttgir_parser_normalized_identical=True,
        source_vs_file_output_bitwise=True,
        input_stage="TTIR after make_ttir; file entry resumes at make_ttgir",
        downstream_stages=list(stages)[1:],
        source=source_info,
        baseline=dump_kernel(baseline, "baseline"),
        toolchain={str(p): sha256(p) for p in tool_paths},
        ptxas_version=subprocess.check_output([ptxas, "--version"], text=True),
        paths={
            name: shutil.which(name)
            for name in ["nvidia-smi", "ptxas", "compute-sanitizer", "llc"]
        },
        environment_variables={
            key: value
            for key, value in os.environ.items()
            if key.startswith(("TRITON", "CUDA", "PYTHON", "LD_", "NVIDIA"))
            or key == "PATH"
        },
        loaded_shared_libraries=libraries,
        gpu_snapshot=snapshot(),
        torch_device_uuid=str(torch.cuda.get_device_properties(0).uuid),
        cuda_stream=torch.cuda.current_stream().cuda_stream,
        git_commit=subprocess.check_output(
            ["git", "rev-parse", "HEAD"], cwd=ROOT, text=True
        ).strip(),
        git_status=subprocess.check_output(
            ["git", "status", "--short"], cwd=ROOT, text=True
        ),
    )
    save_json(args.output, result)
    print(
        "PASS: environment/RECORD; source TTIR; full options/target; "
        "LLIR/PTX/cubin bytes; normalized TTGIR; bitwise output"
    )


if __name__ == "__main__":
    main()
